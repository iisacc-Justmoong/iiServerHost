cmake_minimum_required(VERSION 3.24)
project(iiWebServerHttpConsumer LANGUAGES CXX)
find_package(iiWebServer 0.6 CONFIG REQUIRED COMPONENTS Http)
if(TARGET Qt6::Core OR TARGET iiFileProvider::Native OR TARGET iiWebServer::iiWebServer)
    message(FATAL_ERROR "The Http component must not import compatibility dependencies")
endif()
add_executable(iiWebServerHttpConsumer ../http.cpp)
target_link_libraries(iiWebServerHttpConsumer PRIVATE iiWebServer::Http)
set_target_properties(iiWebServerHttpConsumer PROPERTIES CXX_EXTENSIONS OFF)
enable_testing()
foreach(_case routing protocol lifetime)
    add_test(NAME iiWebServer.http.installed.${_case} COMMAND iiWebServerHttpConsumer ${_case})
    set_tests_properties(iiWebServer.http.installed.${_case} PROPERTIES TIMEOUT 20)
endforeach()
